    <!-- Script pour la navigation et les onglets -->
    <script>
        const navLinks = document.querySelectorAll('.nav-link');
        const pageSections = document.querySelectorAll('.page-section');

        // Fonction pour afficher une section et masquer les autres
        const showSection = (id) => {
            pageSections.forEach(section => {
                section.classList.remove('active');
            });
            const activeSection = document.querySelector(id);
            if (activeSection) {
                activeSection.classList.add('active');
            }
            // Met à jour la classe 'active' pour les liens de navigation
            navLinks.forEach(link => {
                link.classList.remove('active');
                if (link.getAttribute('href') === id) {
                    link.classList.add('active');
                }
            });
        };

        // Gère le clic sur les liens de navigation
        navLinks.forEach(link => {
            link.addEventListener('click', (e) => {
                e.preventDefault();
                const targetId = link.getAttribute('href');
                showSection(targetId);
            });
        });

        // Gère le chargement initial de la page en fonction du hachage de l'URL
        window.addEventListener('load', () => {
            const initialHash = window.location.hash || '#accueil';
            showSection(initialHash);
        });

        // Gère la soumission du formulaire de contact
        const contactForm = document.getElementById('contact-form');
        const formMessage = document.getElementById('form-message');

        contactForm.addEventListener('submit', (e) => {
            e.preventDefault();
            // Simule une soumission réussie
            formMessage.textContent = 'Merci pour votre message ! Il a été envoyé avec succès.';
            formMessage.classList.remove('hidden', 'text-red-500');
            formMessage.classList.add('text-green-500');
            contactForm.reset();
        });
    </script>
